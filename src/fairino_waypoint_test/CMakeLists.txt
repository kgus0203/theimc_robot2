cmake_minimum_required(VERSION 3.8)
project(fairino_waypoint_test)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "action/ScanSide.action")
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(visualization_msgs REQUIRED)
add_executable(waypoint_test src/waypoint_test.cpp)
target_compile_features(waypoint_test PUBLIC cxx_std_17)
target_compile_options(waypoint_test PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(waypoint_test rclcpp rclcpp_action moveit_msgs shape_msgs moveit_ros_planning_interface visualization_msgs)
rosidl_get_typesupport_target(scan_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(waypoint_test "${scan_typesupport}")
ament_export_dependencies(rosidl_default_runtime)
install(TARGETS waypoint_test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
